{{- /* Vykreslí radar graf B612 Compass. Používa ho web aj generátor Instagram postov, takže graf vyzerá na oboch miestach identicky. Farby neurčuje táto šablóna — riadi ich CSS cez triedy `kmp-*`, aby sa graf vedel prispôsobiť krémovému aj nočnému podkladu. Parametre (dict): kompas — mapa {style, story, echo, depth} id — jedinečná predpona pre gradient (viac grafov na stránke) labels — "false" vypne názvy osí font — veľkosť písma popiskov v jednotkách viewBoxu (default 21) */ -}} {{- $b := index site.Data "b612" -}} {{- $k := .kompas -}} {{- $labels := ne .labels "false" -}} {{- $font := .font | default 21 -}} {{- $id := .id | default "b612" -}} {{- $max := float $b.compass.max -}} {{- $cx := 250.0 -}} {{- $cy := 250.0 -}} {{- $R := 140.0 -}} {{- $Rlabel := 196.0 -}} {{- $pts := slice -}} {{- range $b.compass.axes -}} {{- $v := float (index $k .key | default 0) -}} {{- $rad := math.ToRadians (float .angle) -}} {{- $r := mul $R (div $v $max) -}} {{- $pts = $pts | append (printf "%.1f,%.1f" (add $cx (mul $r (math.Cos $rad))) (add $cy (mul $r (math.Sin $rad)))) -}} {{- end -}}
